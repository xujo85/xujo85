        private void video1_MouseDown_1(object sender, MouseEventArgs e)
        {
            if (Player1.IsPlaying)
            {
                Player1.Pause();
            }
            else if (!Player1.IsPlaying)
            {
                Player1.Play();
            }
        }

        private void video1_MouseDoubleClick(object sender, MouseEventArgs e)
        {
            if (!isFullscreen)
            {
                Player1.FullScreen();
                isFullscreen = true;
            }
            else if (isFullscreen)
            {
                Player1.NormalScreen();
                isFullscreen = false;
            }
        }
